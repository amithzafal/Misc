% Best-fit (saturating siRNA) model -- fitted to smFISH mean dh RNA per strain
% (fit.py, results/fit_saturating.json)

\begin{equation}
\frac{dR}{dt}= \frac{T}{1 + (M/K)^k} - \delta_1 R - \gamma R M \,\frac{S}{1 + S/S_p}
\label{R}
\end{equation}

\begin{equation}
\frac{dS}{dt}= \gamma R M \,\frac{S}{1 + S/S_p} + \nu R - \delta_2 S
\label{S}
\end{equation}

\begin{equation}
\frac{dM}{dt}= (1-M)\left[\epsilon_0 R + \left(\epsilon_s + \epsilon_m M\right) R \,\frac{S}{1 + S/S_m}\right] - \delta_3 M
\label{M}
\end{equation}

% S_p, S_m -> infinity recovers the original mass-action model.
% Stochastic version: M = n/N with N nucleosomes; R, S, n integer; 7 reactions
% (transcription, RNA decay, RNA->siRNA, nu R, siRNA decay, methylation, demethylation).
