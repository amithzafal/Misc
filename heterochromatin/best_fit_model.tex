% Best-fit model without RNAi-independent nucleation (epsilon_0 = 0).
% Fitted to smFISH mean dh RNA per strain (fit.py --set eps0=0; results/fit_no_eps0.json).

\begin{equation}
\frac{dR}{dt}= \frac{T}{1 + (M/K)^k} - \delta_1 R - \gamma R M \,\frac{S}{1 + S/S_p}
\label{R}
\end{equation}

\begin{equation}
\frac{dS}{dt}= \gamma R M \,\frac{S}{1 + S/S_p} + \nu R - \delta_2 S
\label{S}
\end{equation}

\begin{equation}
\frac{dM}{dt}= (1-M)\,\left(\epsilon_s + \epsilon_m M\right) R \,\frac{S}{1 + S/S_m} - \delta_3 M
\label{M}
\end{equation}

% All methylation is now RNAi-dependent: it needs both the nascent transcript R
% and siRNA S. Methylation can only start from nu R (Dicer-independent primal
% siRNA). S_p, S_m -> infinity recovers mass-action kinetics.
